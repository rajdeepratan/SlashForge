Plan — per-target command names in the docs

Goal: a reader picks Claude Code, Cursor or Codex once, and every switchable command name on the docs site — code blocks, sentences, page titles and sidebar labels — shows the form that works in their tool.

Architecture

One shared module owns the command list and the three spellings. A rehype plugin wraps every matching reference in <span data-cmd="name"> at build time, already rendered in the Claude Code form, so the page is correct before any script runs. A client script rewrites those spans when the reader picks a target and stores the choice.

Two controls, kept in sync from the same state: a tab strip in each command block's existing bar, and a persistent switcher in the site header for the seven pages that mention commands but contain no code block.

Tech stack

Astro 7 with Shiki, plain ESM plugins over hast (unist-util-visit), framework-free client JavaScript, and the repo's existing .github/scripts/check-docs-*.mjs pattern for build-output assertions. Root tests are node --test with node:assert.

Global constraints

  • The shared module must be .mjs, not .ts. astro.config.mjs imports the rehype plugin in Node during the build, where TypeScript cannot be imported; Vite resolves .mjs for the client bundle. One file has to serve both.
  • The default rendered output must not change. With no script running, every page must read exactly as it does today — the Claude Code form everywhere.
  • setup is not switchable. It has no Cursor or Codex form; deriving one would print a command that does not exist. It stays out of the command list entirely.
  • changelog.md is excluded. Generated from CHANGELOG.md and a historical record; rewriting a v4.0.0 entry would misstate what shipped.
  • Storage key is sf-target, matching the existing sf-theme convention in src/scripts/site.js:9. Every read and write wrapped in try/catch — a private window makes the accessor itself throw.
  • Do not run npm install in docs/. Regenerating the lockfile on macOS drops Linux-only optional dependencies and breaks CI's npm ci. No dependency changes are needed here.
  • Run npm test from the repo root; docs checks run from docs/.

    /**
 * The command names the docs can switch between targets, and the three
 * spellings of each. Imported by the build (astro.config.mjs -> the rehype
 * plugin, in Node) and by the client bundle (src/scripts/site.js, via Vite),
 * which is why this is .mjs and not .ts.
 *
 * The same transformation lives in bin/install.js as toSkillCommandRefs: the
 * installer rewrites cross-references when writing the agents target. Keeping
 * one rule in two places is deliberate — the packages ship separately — but
 * they must agree, and a test asserts the command list does.
 */
export const TARGETS = ['claude', 'cursor', 'codex'];
export const DEFAULT_TARGET = 'claude';
export const STORAGE_KEY = 'sf-target';

// setup is absent here on purpose. It is not installed on cursor or codex, so
// there is no form to switch to; deriving one would print a command that does
// not exist. Every reference to it stays in the Claude Code spelling on every
// target — visibly wrong rather than plausibly wrong.
export const SWITCHABLE = [
  'code', 'investigate', 'review-pr',
  'brainstorm', 'plan', 'debug', 'tdd', 'verify',
  'review-feedback', 'request-review', 'worktree', 'parallel',
];

export function commandForm(name, target) {
  if (target === 'cursor') return `/slashforge-${name}`;
  if (target === 'codex') return `$slashforge-${name}`;
  return `/slashforge:${name}`;
}

// Longest first, so `review-pr` can never win against `review-feedback`.
const alternation = [...SWITCHABLE].sort((a, b) => b.length - a.length).join('|');

export const COMMAND_RE = new RegExp(`/slashforge:(${alternation})\\b`, 'g');

/** The command name when a label is exactly one command, else null. */
export function wholeLabelCommand(label) {
  const re = new RegExp(`^/slashforge:(${alternation})$`);
  const m = re.exec(String(label).trim());
  return m ? m[1] : null;
}

    #!/usr/bin/env node
/** Asserts the per-target command markup is present and correct in dist/. */
import { readFileSync, readdirSync, statSync } from 'node:fs';
import { join } from 'node:path';
import { SWITCHABLE } from '../../docs/src/targets.mjs';

const base = process.env.DOCS_BASE_PATH ?? '/slashforge';
const root = join(process.cwd(), 'dist' + base);

const pages = [];
(function walk(d) {
  for (const e of readdirSync(d)) {
    const p = join(d, e);
    if (statSync(p).isDirectory()) walk(p);
    else if (e === 'index.html') pages.push(p);
  }
})(root);

let bad = 0;
const fail = (m) => { console.error('  ✗ ' + m); bad += 1; };

for (const p of pages) {
  const html = readFileSync(p, 'utf8');
  const rel = p.slice(root.length);

  // Every marked span ships the Claude Code spelling.
  for (const m of html.matchAll(/<span data-cmd="([a-z-]+)">([^<]*)<\/span>/g)) {
    if (m[2] !== '/slashforge:' + m[1]) fail(`${rel}: ${m[1]} rendered as "${m[2]}"`);
    if (!SWITCHABLE.includes(m[1])) fail(`${rel}: marked a non-switchable command "${m[1]}"`);
  }

  // setup must never be marked.
  if (/data-cmd="setup"/.test(html)) fail(`${rel}: setup must not be switchable`);

  // The changelog is a historical record and is left alone.
  if (rel.startsWith('/changelog') && /data-cmd=/.test(html)) {
    fail('changelog must not be rewritten');
  }

  // A pre containing a marked command must be flagged for the tab strip.
  for (const pre of html.match(/<pre[^>]*>[\s\S]*?<\/pre>/g) ?? []) {
    const has = /data-cmd=/.test(pre);
    const flagged = /data-has-cmd/.test(pre);
    if (has && !flagged) fail(`${rel}: block with commands is missing data-has-cmd`);
    if (!has && flagged) fail(`${rel}: block without commands is flagged`);
  }
}

console.log(`checked ${pages.length} pages`);
if (bad) { console.error(`${bad} problem(s)`); process.exit(1); }
console.log('per-target command markup is correct');

  • ☐ Step 3: Write the plugin
    One pass splits matching text nodes; a second flags the enclosing blocks. The text pass covers inline <code> and fenced blocks alike, because both are text nodes by the time rehype runs — Shiki emits one text node per line for plaintext blocks, and all 14 command-bearing fences in the site are plaintext. 
    import { visit } from 'unist-util-visit';
import { COMMAND_RE, commandForm, DEFAULT_TARGET } from '../targets.mjs';

export function rehypeCommandNames() {
  return (tree, file) => {
    // The changelog is generated from /CHANGELOG.md and is a historical
    // record: rewriting a v4.0.0 entry would misstate what shipped.
    const path = String(file?.path ?? file?.history?.[0] ?? '');
    if (path.endsWith('changelog.md')) return;

    visit(tree, 'text', (node, index, parent) => {
      if (!parent || index === null) return;
      COMMAND_RE.lastIndex = 0;
      if (!COMMAND_RE.test(node.value)) return;

      const out = [];
      let last = 0;
      let m;
      COMMAND_RE.lastIndex = 0;
      while ((m = COMMAND_RE.exec(node.value)) !== null) {
        if (m.index > last) {
          out.push({ type: 'text', value: node.value.slice(last, m.index) });
        }
        out.push({
          type: 'element',
          tagName: 'span',
          properties: { 'data-cmd': m[1] },
          children: [{ type: 'text', value: commandForm(m[1], DEFAULT_TARGET) }],
        });
        last = m.index + m[0].length;
      }
      if (last < node.value.length) {
        out.push({ type: 'text', value: node.value.slice(last) });
      }

      parent.children.splice(index, 1, ...out);
      // Continue after what was inserted, or visit() walks the new spans.
      return index + out.length;
    });

    visit(tree, 'element', (node) => {
      if (node.tagName !== 'pre') return;
      let has = false;
      visit(node, 'element', (n) => {
        if (n.properties && n.properties['data-cmd'] !== undefined) has = true;
      });
      if (has) node.properties = { ...(node.properties ?? {}), 'data-has-cmd': '' };
    });
  };
}

  • ☐ Step 1: Add the import and the block
    Follows the existing theme code exactly: same storage convention, same try/catch, same attribute-on-root shape. 
    import { TARGETS, DEFAULT_TARGET, STORAGE_KEY, commandForm } from '../targets.mjs';

/* ---- coding-agent target ---- */
(function () {
  var root = document.documentElement;

  function read() {
    try {
      var v = localStorage.getItem(STORAGE_KEY);
      return TARGETS.indexOf(v) >= 0 ? v : DEFAULT_TARGET;
    } catch (e) {
      return DEFAULT_TARGET;
    }
  }

  function apply(t) {
    root.setAttribute('data-target', t);
    document.querySelectorAll('[data-cmd]').forEach(function (el) {
      el.textContent = commandForm(el.getAttribute('data-cmd'), t);
    });
    document.querySelectorAll('[data-target-opt]').forEach(function (b) {
      b.setAttribute('aria-selected', String(b.getAttribute('data-target-opt') === t));
    });
  }

  function set(t) {
    apply(t);
    try { localStorage.setItem(STORAGE_KEY, t); } catch (e) {}
  }

  apply(read());

  document.addEventListener('click', function (e) {
    var btn = e.target.closest('[data-target-opt]');
    if (btn) set(btn.getAttribute('data-target-opt'));
  });

  // Arrow keys move between tabs, as the tablist pattern expects.
  document.addEventListener('keydown', function (e) {
    var btn = e.target.closest('[data-target-opt]');
    if (!btn) return;
    var i = TARGETS.indexOf(btn.getAttribute('data-target-opt'));
    var next = e.key === 'ArrowRight' ? i + 1 : e.key === 'ArrowLeft' ? i - 1 : -1;
    if (next < 0 || next >= TARGETS.length) return;
    e.preventDefault();
    set(TARGETS[next]);
    var sel = btn.parentNode.querySelector('[data-target-opt="' + TARGETS[next] + '"]');
    if (sel) sel.focus();
  });
})();
