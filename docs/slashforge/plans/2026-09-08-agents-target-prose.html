Plan — target-aware workflow prose for Cursor and Codex

Goal: the twelve commands installed on the Cursor/Codex target stop instructing the model to dispatch subagents and create .claude/agents/ files, while Claude Code's installed output stays exactly as it is today.

Architecture

Guides and commands stay single files. Passages that differ per target are fenced with HTML comments — <!--target:claude--> … <!--/target--> — and a new stripTargetBlocks() keeps the block matching the target being written and deletes the rest. It is called from renderTemplate(), which is the one function both the guide loop and the command loop already pass every file through, so a single insertion point covers all twenty-nine files.

Malformed markers abort the install at validation time rather than silently mis-stripping a guide. A guide that loses half a sentence is worse than a failed install, because the result is prose a model then follows.

Tech stack

Node.js CommonJS, no dependencies. Tests are node --test with node:assert. Templates are Markdown with YAML frontmatter.

Global constraints

  • Node 16 is the floor. CI runs the matrix 16, 18, 20, 22 (.github/workflows/ci.yml). No syntax or API newer than Node 16 — structuredClone and the regex v flag are out.
  • Zero runtime dependencies. The published package ships bin/ and templates/ only. Do not add a parser or a templating library.
  • Claude Code output must not change. Every task that touches a template must leave the claude-target render of that file byte-for-byte as it is today apart from removed markers.
  • Valid target names are exactly claude and agents. cursor and codex are aliases resolved before this point and must never appear in a marker.
  • The five existing tests asserting setup is omitted on the agents target must keep passing untouched (test/install.test.js lines 834, 855, 863, 987, 1043).
  • Run npm test from the repo root. There is no lint or build step for the package itself.

    // Passages that differ between install targets are fenced in the templates:
//   <!--target:claude--> ... <!--/target-->
// Everything unfenced is shared. Keeping one source per file is what stops the
// two variants drifting; the alternative was a near-duplicate of every guide.
const TARGET_BLOCK_RE =
  /^[ \t]*<!--target:([a-z-]+)-->[ \t]*\n([\s\S]*?)^[ \t]*<!--\/target-->[ \t]*\n?/gm;

function stripTargetBlocks(content, targetName) {
  return content.replace(TARGET_BLOCK_RE, (_match, name, body) =>
    (name === targetName ? body : '')
  );
}

    const VALID_TARGET_NAMES = ['claude', 'agents'];
const TARGET_TOKEN_RE = /<!--target:([a-z-]+)-->|<!--\/target-->/g;

function findTargetBlockErrors(content, file) {
  const errors = [];
  let open = null;
  let match;
  TARGET_TOKEN_RE.lastIndex = 0;
  while ((match = TARGET_TOKEN_RE.exec(content)) !== null) {
    const name = match[1];
    const line = content.slice(0, match.index).split('\n').length;
    if (name === undefined) {
      if (open === null) errors.push(`${file}:${line}: <!--/target--> with no open block`);
      else open = null;
      continue;
    }
    if (open !== null) {
      errors.push(`${file}:${line}: nested target block (inside '${open}')`);
      continue;
    }
    if (!VALID_TARGET_NAMES.includes(name)) {
      errors.push(
        `${file}:${line}: unknown target '${name}' — use ${VALID_TARGET_NAMES.join(' or ')}`
      );
    }
    open = name;
  }
  if (open !== null) errors.push(`${file}: unclosed target block '${open}'`);
  return errors;
}

    const { renderTemplate } = require('../bin/install.js');
const TEMPLATES = path.join(__dirname, '..', 'templates');

function renderAll(targetName) {
  const out = {};
  const walk = (dir, rel = '') => {
    for (const e of fs.readdirSync(dir, { withFileTypes: true })) {
      const full = path.join(dir, e.name);
      if (e.isDirectory()) walk(full, path.join(rel, e.name));
      else if (e.name.endsWith('.md')) {
        out[path.join(rel, e.name)] = renderTemplate(fs.readFileSync(full, 'utf8'), {
          installPath: '/p', version: '0.0.0', pkgName: 'slashforge', targetName,
        });
      }
    }
  };
  walk(TEMPLATES);
  return out;
}

test('no target marker survives into any rendered file', () => {
  for (const targetName of ['claude', 'agents']) {
    for (const [file, body] of Object.entries(renderAll(targetName))) {
      assert.ok(!body.includes('<!--target:'), `${file} kept an open marker on ${targetName}`);
      assert.ok(!body.includes('<!--/target-->'), `${file} kept a close marker on ${targetName}`);
    }
  }
});

test('renderTemplate strips target blocks for the given target', () => {
  const src = 'a\n<!--target:claude-->\nC\n<!--/target-->\n<!--target:agents-->\nA\n<!--/target-->\n';
  const opts = { installPath: '/p', version: '0.0.0', pkgName: 'slashforge' };
  assert.equal(renderTemplate(src, { ...opts, targetName: 'claude' }), 'a\nC\n');
  assert.equal(renderTemplate(src, { ...opts, targetName: 'agents' }), 'a\nA\n');

test('rendering only removes whole lines, never rewrites them', () => {
  for (const targetName of ['claude', 'agents']) {
    for (const [file, body] of Object.entries(renderAll(targetName))) {
      const src = fs.readFileSync(path.join(TEMPLATES, file), 'utf8')
        .replace(/\{\{INSTALL_PATH\}\}/g, '/p')
        .replace(/\{\{KIT_VERSION\}\}/g, '0.0.0')
        .replace(/\{\{KIT_PACKAGE\}\}/g, 'slashforge')
        .split('\n');
      let i = 0;
      for (const line of body.split('\n')) {
        while (i < src.length && src[i] !== line) i += 1;
        assert.ok(i < src.length, `${file} (${targetName}): rendered line not in source: ${line}`);
        i += 1;
      }
    }
  }
});
